### 3.2 Компоненты для улучшения UX

```javascript
// ux-components.js - UX компоненты для валидации

/**
 * Класс для создания интерактивных компонентов UX
 */
class UXComponents {
  constructor() {
    this.components = new Map();
    this.observers = new Map();
  }

  /**
   * Создать индикатор загрузки для асинхронной валидации
   */
  createLoadingIndicator(fieldName) {
    const field = document.getElementById(fieldName);
    if (!field) return null;

    const indicator = document.createElement("div");
    indicator.className = "validation-loading";
    indicator.innerHTML = `
            <div class="loading-spinner">
                <svg width="16" height="16" viewBox="0 0 24 24">
                    <circle class="spinner-circle" cx="12" cy="12" r="10" 
                            fill="none" stroke="currentColor" stroke-width="2"/>
                </svg>
            </div>
            <span class="loading-text">Проверка...</span>
        `;

    // Позиционируем индикатор
    const fieldRect = field.getBoundingClientRect();
    indicator.style.position = "absolute";
    indicator.style.top = `${fieldRect.bottom + 5}px`;
    indicator.style.left = `${fieldRect.left}px`;
    indicator.style.zIndex = "1000";

    document.body.appendChild(indicator);
    this.components.set(`${fieldName}-loading`, indicator);

    return indicator;
  }

  /**
   * Удалить индикатор загрузки
   */
  removeLoadingIndicator(fieldName) {
    const indicator = this.components.get(`${fieldName}-loading`);
    if (indicator && indicator.parentNode) {
      indicator.parentNode.removeChild(indicator);
      this.components.delete(`${fieldName}-loading`);
    }
  }

  /**
   * Создать всплывающую подсказку с контекстной помощью
   */
  createContextualHelp(fieldName, helpData) {
    const field = document.getElementById(fieldName);
    if (!field) return null;

    const helpButton = document.createElement("button");
    helpButton.type = "button";
    helpButton.className = "contextual-help-trigger";
    helpButton.setAttribute("aria-label", "Показать справку");
    helpButton.innerHTML = `
            <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor">
                <circle cx="12" cy="12" r="10"/>
                <path d="M9.09 9a3 3 0 0 1 5.83 1c0 2-3 3-3 3"/>
                <line x1="12" y1="17" x2="12.01" y2="17"/>
            </svg>
        `;

    // Вставляем кнопку после поля
    field.parentNode.insertBefore(helpButton, field.nextSibling);

    // Создаем панель помощи
    const helpPanel = document.createElement("div");
    helpPanel.className = "contextual-help-panel";
    helpPanel.innerHTML = `
            <div class="help-content">
                <h4>${helpData.title}</h4>
                <p>${helpData.description}</p>
                ${
                  helpData.examples
                    ? `
                    <div class="help-examples">
                        <strong>Примеры:</strong>
                        <ul>
                            ${helpData.examples
                              .map((ex) => `<li>${ex}</li>`)
                              .join("")}
                        </ul>
                    </div>
                `
                    : ""
                }
                ${
                  helpData.tips
                    ? `
                    <div class="help-tips">
                        <strong>Советы:</strong>
                        <ul>
                            ${helpData.tips
                              .map((tip) => `<li>${tip}</li>`)
                              .join("")}
                        </ul>
                    </div>
                `
                    : ""
                }
            </div>
            <button class="help-close" aria-label="Закрыть справку">&times;</button>
        `;

    document.body.appendChild(helpPanel);

    // Обработчики событий
    helpButton.addEventListener("click", () => {
      this.showContextualHelp(helpPanel, helpButton);
    });

    helpPanel.querySelector(".help-close").addEventListener("click", () => {
      this.hideContextualHelp(helpPanel);
    });

    this.components.set(`${fieldName}-help`, {
      button: helpButton,
      panel: helpPanel,
    });

    return { button: helpButton, panel: helpPanel };
  }

  /**
   * Показать контекстную справку
   */
  showContextualHelp(panel, trigger) {
    // Скрываем все другие панели справки
    document.querySelectorAll(".contextual-help-panel.show").forEach((p) => {
      if (p !== panel) {
        this.hideContextualHelp(p);
      }
    });

    // Позиционируем панель
    const triggerRect = trigger.getBoundingClientRect();
    panel.style.position = "fixed";
    panel.style.top = `${triggerRect.bottom + 10}px`;
    panel.style.left = `${triggerRect.left}px`;
    panel.style.zIndex = "10000";

    // Проверяем границы экрана
    const panelRect = panel.getBoundingClientRect();
    if (panelRect.right > window.innerWidth - 20) {
      panel.style.left = `${window.innerWidth - panelRect.width - 20}px`;
    }
    if (panelRect.bottom > window.innerHeight - 20) {
      panel.style.top = `${triggerRect.top - panelRect.height - 10}px`;
    }

    // Показываем с анимацией
    panel.classList.add("show");
  }

  /**
   * Скрыть контекстную справку
   */
  hideContextualHelp(panel) {
    panel.classList.remove("show");
  }

  /**
   * Создать прогресс-бар для многошаговой формы
   */
  createStepProgress(steps, currentStep = 0) {
    const progressContainer = document.createElement("div");
    progressContainer.className = "step-progress";

    steps.forEach((step, index) => {
      const stepElement = document.createElement("div");
      stepElement.className = "step";
      stepElement.setAttribute("data-step", index);
      stepElement.textContent = index + 1;

      if (index < currentStep) {
        stepElement.classList.add("completed");
      } else if (index === currentStep) {
        stepElement.classList.add("current");
      }

      // Добавляем подпись к шагу
      const stepLabel = document.createElement("div");
      stepLabel.className = "step-label";
      stepLabel.textContent = step.label;

      const stepWrapper = document.createElement("div");
      stepWrapper.className = "step-wrapper";
      stepWrapper.appendChild(stepElement);
      stepWrapper.appendChild(stepLabel);

      progressContainer.appendChild(stepWrapper);
    });

    return progressContainer;
  }

  /**
   * Обновить прогресс-бар
   */
  updateStepProgress(progressContainer, currentStep) {
    const steps = progressContainer.querySelectorAll(".step");

    steps.forEach((step, index) => {
      step.classList.remove("current", "completed");

      if (index < currentStep) {
        step.classList.add("completed");
      } else if (index === currentStep) {
        step.classList.add("current");
      }
    });
  }

  /**
   * Создать live-регион для объявлений скринридерам
   */
  createLiveRegion() {
    const liveRegion = document.createElement("div");
    liveRegion.setAttribute("aria-live", "polite");
    liveRegion.setAttribute("aria-atomic", "true");
    liveRegion.className = "sr-only";
    liveRegion.id = "validation-live-region";

    document.body.appendChild(liveRegion);
    this.components.set("live-region", liveRegion);

    return liveRegion;
  }

  /**
   * Объявить сообщение в live-регионе
   */
  announceToScreenReader(message) {
    let liveRegion = this.components.get("live-region");
    if (!liveRegion) {
      liveRegion = this.createLiveRegion();
    }

    liveRegion.textContent = message;

    // Очищаем через короткое время, чтобы можно было объявить то же сообщение снова
    setTimeout(() => {
      liveRegion.textContent = "";
    }, 1000);
  }

  /**
   * Создать tooltip с дополнительной информацией
   */
  createTooltip(trigger, content, options = {}) {
    const tooltip = document.createElement("div");
    tooltip.className = "validation-tooltip";
    tooltip.innerHTML = content;
    tooltip.setAttribute("role", "tooltip");

    const settings = {
      placement: "top",
      delay: 300,
      duration: 3000,
      ...options,
    };

    let showTimeout, hideTimeout;

    const showTooltip = () => {
      clearTimeout(hideTimeout);
      showTimeout = setTimeout(() => {
        document.body.appendChild(tooltip);
        this.positionTooltip(tooltip, trigger, settings.placement);
        tooltip.classList.add("show");

        if (settings.duration > 0) {
          hideTimeout = setTimeout(() => {
            this.hideTooltip(tooltip);
          }, settings.duration);
        }
      }, settings.delay);
    };

    const hideTooltip = () => {
      clearTimeout(showTimeout);
      this.hideTooltip(tooltip);
    };

    // Обработчики событий
    trigger.addEventListener("mouseenter", showTooltip);
    trigger.addEventListener("mouseleave", hideTooltip);
    trigger.addEventListener("focus", showTooltip);
    trigger.addEventListener("blur", hideTooltip);

    return tooltip;
  }

  /**
   * Позиционировать tooltip
   */
  positionTooltip(tooltip, trigger, placement) {
    const triggerRect = trigger.getBoundingClientRect();
    const tooltipRect = tooltip.getBoundingClientRect();
    const scrollTop = window.pageYOffset || document.documentElement.scrollTop;
    const scrollLeft =
      window.pageXOffset || document.documentElement.scrollLeft;

    let top, left;

    switch (placement) {
      case "top":
        top = triggerRect.top + scrollTop - tooltipRect.height - 10;
        left =
          triggerRect.left +
          scrollLeft +
          (triggerRect.width - tooltipRect.width) / 2;
        break;
      case "bottom":
        top = triggerRect.bottom + scrollTop + 10;
        left =
          triggerRect.left +
          scrollLeft +
          (triggerRect.width - tooltipRect.width) / 2;
        break;
      case "left":
        top =
          triggerRect.top +
          scrollTop +
          (triggerRect.height - tooltipRect.height) / 2;
        left = triggerRect.left + scrollLeft - tooltipRect.width - 10;
        break;
      case "right":
        top =
          triggerRect.top +
          scrollTop +
          (triggerRect.height - tooltipRect.height) / 2;
        left = triggerRect.right + scrollLeft + 10;
        break;
    }

    // Проверяем границы экрана
    const padding = 10;
    left = Math.max(
      padding,
      Math.min(left, window.innerWidth - tooltipRect.width - padding)
    );
    top = Math.max(
      padding,
      Math.min(top, window.innerHeight - tooltipRect.height - padding)
    );

    tooltip.style.position = "absolute";
    tooltip.style.top = `${top}px`;
    tooltip.style.left = `${left}px`;
    tooltip.style.zIndex = "10000";
  }

  /**
   * Скрыть tooltip
   */
  hideTooltip(tooltip) {
    tooltip.classList.remove("show");
    setTimeout(() => {
      if (tooltip.parentNode) {
        tooltip.parentNode.removeChild(tooltip);
      }
    }, 300);
  }

  /**
   * Очистить все компоненты
   */
  cleanup() {
    this.components.forEach((component, key) => {
      if (component.parentNode) {
        component.parentNode.removeChild(component);
      }
    });
    this.components.clear();

    this.observers.forEach((observer) => {
      observer.disconnect();
    });
    this.observers.clear();
  }
}
```

```javascript
// main-validation-example.js - Пример полной интеграции

document.addEventListener("DOMContentLoaded", function () {

    uxComponents.announceToScreenReader("Все поля заполнены корректно");
    console.log("Form is ready to submit:", validationResult);

    // Здесь можно отправить форму на сервер
    submitForm(form);
  };

  validator.onFormInvalid = function (form, validationResult) {
    const errorCount = Object.values(validationResult.results).filter(
      (result) => !result.isValid
    ).length;

    uxComponents.announceToScreenReader(
      `Найдено ошибок: ${errorCount}. Исправьте их для продолжения.`

    const buttonText = submitButton.querySelector(".btn-text");
    const buttonLoader = submitButton.querySelector(".btn-loader");

    try {
      // Показываем состояние загрузки
      buttonText.style.display = "none";
      buttonLoader.style.display = "block";
      submitButton.disabled = true;

      // Отправляем данные

      const response = await fetch("/api/auth/register", {
        method: "POST",
        body: formData,

        throw new Error(errorData.message || "Ошибка регистрации");

      uxComponents.announceToScreenReader(`Ошибка: ${error.message}`);

      showErrorNotification(error.message);
    } finally {

      buttonText.style.display = "block";
      buttonLoader.style.display = "none";
      submitButton.disabled = false;
    }
  }

  function showErrorNotification(message) {
    // Создаем уведомление об ошибке
    const notification = document.createElement("div");
    notification.className = "notification error";
    notification.textContent = message;

    document.body.appendChild(notification);

    setTimeout(() => {
      notification.classList.add("show");
    }, 100);

    setTimeout(() => {
      notification.classList.remove("show");
      setTimeout(() => {
        if (notification.parentNode) {
          notification.parentNode.removeChild(notification);
        }
      }, 300);
    }, 5000);
  }
});
```

### Пример 2: Прогрессивная валидация для многошаговой формы

```javascript
// progressive-form-example.js - Пример прогрессивной формы

document.addEventListener("DOMContentLoaded", function () {
  const progressiveValidator = new ProgressiveValidator();
  const uxComponents = new UXComponents();

  // Определяем шаги валидации
  const validationSteps = [
    {
      label: "Личные данные",
      fields: [
        { name: "firstName", validator: "name" },
        { name: "lastName", validator: "name" },
      ],
    },
    {
      label: "Учетные данные",
      fields: [
        { name: "email", validator: "emailWithAvailability" },
        { name: "username", validator: "usernameWithAvailability" },
      ],
    },
    {
      label: "Безопасность",
      fields: [
        { name: "password", validator: "passwordStrength" },
        { name: "confirmPassword", validator: "confirmPassword" },
      ],
    },
  ];

  // Инициализируем шаги
  progressiveValidator.initValidationSteps(validationSteps);

  // Создаем прогресс-бар
  const progressBar = uxComponents.createStepProgress(validationSteps);
  const formContainer = document.querySelector(".form-container");
  formContainer.insertBefore(progressBar, formContainer.firstChild);

  // Инициализируем валидацию формы
  progressiveValidator.initFormValidation("#multiStepForm");

  // Обработчики для кнопок навигации
  const nextButton = document.getElementById("nextStep");
  const prevButton = document.getElementById("prevStep");
  const submitButton = document.getElementById("submitForm");

  if (nextButton) {
    nextButton.addEventListener("click", async function () {
      const form = document.getElementById("multiStepForm");
      const result = await progressiveValidator.nextStep(form);

      if (result.success) {
        uxComponents.updateStepProgress(progressBar, result.nextStep);
        updateStepVisibility(result.nextStep);
        uxComponents.announceToScreenReader(
          `Переход к шагу ${result.nextStep + 1}`
        );
      } else {
        uxComponents.announceToScreenReader("Исправьте ошибки для продолжения");
      }
    });
  }

  if (prevButton) {
    prevButton.addEventListener("click", function () {
      const result = progressiveValidator.previousStep();

      if (result.success) {
        uxComponents.updateStepProgress(progressBar, result.previousStep);
        updateStepVisibility(result.previousStep);
        uxComponents.announceToScreenReader(
          `Возврат к шагу ${result.previousStep + 1}`
        );
      }
    });
  }

  function updateStepVisibility(currentStep) {
    const steps = document.querySelectorAll("[data-step]");

    steps.forEach((step, index) => {
      step.style.display = index === currentStep ? "block" : "none";
    });

    // Обновляем видимость кнопок
    if (prevButton) {
      prevButton.style.display = currentStep > 0 ? "block" : "none";
    }

    if (nextButton) {
      nextButton.style.display =
        currentStep < validationSteps.length - 1 ? "block" : "none";
    }

    if (submitButton) {
      submitButton.style.display =
        currentStep === validationSteps.length - 1 ? "block" : "none";
    }
  }

  // Инициализируем видимость
  updateStepVisibility(0);
});
```
