```javascript
// auth-storage.js - Система хранения

/**
 * Класс для безопасного хранения данных авторизации
 */
class AuthStorage {
  constructor(config) {
    this.config = config;
    this.encryptionKey = this.generateEncryptionKey();
    this.storage = window.localStorage;
    this.sessionStorage = window.sessionStorage;
  }

  /**
   * Инициализация системы хранения
   */
  async init() {
    // Проверяем доступность localStorage
    if (!this.isStorageAvailable()) {
      console.warn(
        "localStorage not available, falling back to memory storage"
      );
      this.initMemoryStorage();
    }

    // Очищаем устаревшие данные
    this.cleanupExpiredData();

    console.log("AuthStorage initialized");
  }

  /**
   * Проверка доступности localStorage
   */
  isStorageAvailable() {
    try {
      const test = "__storage_test__";
      this.storage.setItem(test, test);
      this.storage.removeItem(test);
      return true;
    } catch (e) {
      return false;
    }
  }

  /**
   * Инициализация хранения в памяти (fallback)
   */
  initMemoryStorage() {
    this.memoryStorage = new Map();

    this.storage = {
      getItem: (key) => this.memoryStorage.get(key) || null,
      setItem: (key, value) => this.memoryStorage.set(key, value),
      removeItem: (key) => this.memoryStorage.delete(key),
      clear: () => this.memoryStorage.clear(),
    };
  }

  /**
   * Сохранение данных авторизации
   */
  saveAuth(authData, remember = false) {
    const storage = remember ? this.storage : this.sessionStorage;

    try {
      // Сохраняем токен
      if (authData.token) {
        this.setSecure("authToken", authData.token, storage);
      }

      // Сохраняем refresh token
      if (authData.refreshToken) {
        this.setSecure("refreshToken", authData.refreshToken, storage);
      }

      // Сохраняем данные пользователя
      if (authData.user) {
        this.setSecure("currentUser", JSON.stringify(authData.user), storage);
      }

      // Сохраняем время истечения
      if (authData.expiresIn) {
        const expirationTime = Date.now() + authData.expiresIn * 1000;
        storage.setItem("tokenExpiration", expirationTime.toString());
      }

      // Сохраняем метаданные
      const metadata = {
        loginTime: Date.now(),
        remember: remember,
        version: "1.0",
      };
      storage.setItem("authMetadata", JSON.stringify(metadata));

      console.log("Auth data saved successfully");
    } catch (error) {
      console.error("Failed to save auth data:", error);
      throw new Error("Storage error");
    }
  }

  /**
   * Получение токена
   */
  getToken() {
    return (
      this.getSecure("authToken") ||
      this.getSecure("authToken", this.sessionStorage)
    );
  }

  /**
   * Получение refresh token
   */
  getRefreshToken() {
    return (
      this.getSecure("refreshToken") ||
      this.getSecure("refreshToken", this.sessionStorage)
    );
  }

  /**
   * Получение данных пользователя
   */
  getUser() {
    const userData =
      this.getSecure("currentUser") ||
      this.getSecure("currentUser", this.sessionStorage);

    if (userData) {
      try {
        return JSON.parse(userData);
      } catch (error) {
        console.error("Failed to parse user data:", error);
        return null;
      }
    }

    return null;
  }

  /**
   * Получение времени истечения токена
   */
  getTokenExpiration() {
    const expiration =
      this.storage.getItem("tokenExpiration") ||
      this.sessionStorage.getItem("tokenExpiration");

    return expiration ? parseInt(expiration, 10) : null;
  }

  /**
   * Получение метаданных авторизации
   */
  getAuthMetadata() {
    const metadata =
      this.storage.getItem("authMetadata") ||
      this.sessionStorage.getItem("authMetadata");

    if (metadata) {
      try {
        return JSON.parse(metadata);
      } catch (error) {
        console.error("Failed to parse auth metadata:", error);
        return null;
      }
    }

    return null;
  }

  /**
   * Очистка всех данных авторизации
   */
  clearAuth() {
    const keys = [
      "authToken",
      "refreshToken",
      "currentUser",
      "tokenExpiration",
      "authMetadata",
    ];

    keys.forEach((key) => {
      this.removeSecure(key);
      this.removeSecure(key, this.sessionStorage);
      this.storage.removeItem(key);
      this.sessionStorage.removeItem(key);
    });

    console.log("Auth data cleared");
  }

  /**
   * Безопасное сохранение данных с шифрованием
   */
  setSecure(key, value, storage = this.storage) {
    try {
      const encrypted = this.encrypt(value);
      storage.setItem(`secure_${key}`, encrypted);
    } catch (error) {
      console.error("Failed to encrypt data:", error);
      // Fallback к обычному сохранению
      storage.setItem(key, value);
    }
  }

  /**
   * Безопасное получение зашифрованных данных
   */
  getSecure(key, storage = this.storage) {
    try {
      const encrypted = storage.getItem(`secure_${key}`);
      if (encrypted) {
        return this.decrypt(encrypted);
      }

      // Fallback к обычному получению
      return storage.getItem(key);
    } catch (error) {
      console.error("Failed to decrypt data:", error);
      return storage.getItem(key);
    }
  }

  /**
   * Удаление зашифрованных данных
   */
  removeSecure(key, storage = this.storage) {
    storage.removeItem(`secure_${key}`);
    storage.removeItem(key);
  }

  /**
   * Простое шифрование (для демонстрации)
   */
  encrypt(text) {
    if (!text) return text;

    // Используем простой XOR шифр с ключом
    let result = "";
    for (let i = 0; i < text.length; i++) {
      result += String.fromCharCode(
        text.charCodeAt(i) ^
          this.encryptionKey.charCodeAt(i % this.encryptionKey.length)
      );
    }

    return btoa(result);
  }

  /**
   * Расшифровка
   */
  decrypt(encryptedText) {
    if (!encryptedText) return encryptedText;

    try {
      const text = atob(encryptedText);
      let result = "";

      for (let i = 0; i < text.length; i++) {
        result += String.fromCharCode(
          text.charCodeAt(i) ^
            this.encryptionKey.charCodeAt(i % this.encryptionKey.length)
        );
      }

      return result;
    } catch (error) {
      console.error("Decryption failed:", error);
      return encryptedText;
    }
  }

  /**
   * Генерация ключа шифрования
   */
  generateEncryptionKey() {
    // В реальном приложении используйте более безопасный метод
    return "AuthKey_" + (navigator.userAgent || "default").slice(-20);
  }

  /**
   * Очистка устаревших данных
   */
  cleanupExpiredData() {
    const expiration = this.getTokenExpiration();

    if (expiration && expiration < Date.now()) {
      console.log("Clearing expired auth data");
      this.clearAuth();
    }
  }

  /**
   * Получение размера занятого места
   */
  getStorageSize() {
    let total = 0;

    for (let key in this.storage) {
      if (this.storage.hasOwnProperty(key)) {
        total += this.storage[key].length + key.length;
      }
    }

    return total;
  }

  /**
   * Экспорт данных для резервного копирования
   */
  exportData() {
    const data = {
      user: this.getUser(),
      metadata: this.getAuthMetadata(),
      timestamp: Date.now(),
    };

    return JSON.stringify(data);
  }

  /**
   * Импорт данных из резервной копии
   */
  importData(dataString) {
    try {
      const data = JSON.parse(dataString);

      if (data.user) {
        this.setSecure("currentUser", JSON.stringify(data.user));
      }

      if (data.metadata) {
        this.storage.setItem("authMetadata", JSON.stringify(data.metadata));
      }

      console.log("Data imported successfully");
      return true;
    } catch (error) {
      console.error("Failed to import data:", error);
      return false;
    }
  }
}
```

```javascript
// auth-router.js - Роутер с защитой маршрутов

/**
 * Простой роутер с поддержкой защищенных маршрутов
 */
class AuthRouter {
  constructor(eventBus) {
    this.eventBus = eventBus;
    this.routes = new Map();
    this.guards = new Map();
    this.currentRoute = null;
    this.isNavigating = false;

    this.setupDefaultRoutes();
  }

  /**
   * Инициализация роутера
   */
  async init() {
    // Слушаем изменения URL
    window.addEventListener("popstate", this.handlePopState.bind(this));

    // Обрабатываем текущий маршрут
    await this.handleCurrentRoute();

    console.log("AuthRouter initialized");
  }

  /**
   * Настройка стандартных маршрутов
   */
  setupDefaultRoutes() {
    // Публичные маршруты
    this.addRoute("/login", {
      public: true,
      component: "LoginPage",
      title: "Вход в систему",
    });

    this.addRoute("/register", {
      public: true,
      component: "RegisterPage",
      title: "Регистрация",
    });

    this.addRoute("/forgot-password", {
      public: true,
      component: "ForgotPasswordPage",
      title: "Восстановление пароля",
    });

    // Защищенные маршруты
    this.addRoute("/dashboard", {
      protected: true,
      component: "DashboardPage",
      title: "Личный кабинет",
    });

    this.addRoute("/profile", {
      protected: true,
      component: "ProfilePage",
      title: "Профиль пользователя",
    });

    this.addRoute("/admin", {
      protected: true,
      roles: ["admin"],
      component: "AdminPage",
      title: "Панель администратора",
    });

    // Маршруты ошибок
    this.addRoute("/403", {
      public: true,
      component: "ForbiddenPage",
      title: "Доступ запрещен",
    });

    this.addRoute("/404", {
      public: true,
      component: "NotFoundPage",
      title: "Страница не найдена",
    });
  }

  /**
   * Добавление маршрута
   */
  addRoute(path, config) {
    this.routes.set(path, {
      path,
      public: false,
      protected: false,
      roles: [],
      permissions: [],
      component: null,
      title: "",
      meta: {},
      ...config,
    });
  }

  /**
   * Добавление guard'а для защиты маршрутов
   */
  addGuard(name, guardFunction) {
    this.guards.set(name, guardFunction);
  }

  /**
   * Навигация к маршруту
   */
  async navigate(path, options = {}) {
    if (this.isNavigating) {
      console.warn("Navigation already in progress");
      return false;
    }

    this.isNavigating = true;

    try {
      const route = this.findRoute(path);

      if (!route) {
        return this.navigate("/404");
      }

      // Проверяем доступ к маршруту
      const canAccess = await this.checkAccess(route);

      if (!canAccess.allowed) {
        if (canAccess.redirectTo) {
          return this.navigate(canAccess.redirectTo);
        } else {
          return this.navigate("/403");
        }
      }

      // Уведомляем о начале навигации
      this.eventBus.emit("route:before", {
        from: this.currentRoute,
        to: route,
      });

      // Обновляем URL если необходимо
      if (!options.replace) {
        history.pushState({ path }, route.title, path);
      } else {
        history.replaceState({ path }, route.title, path);
      }

      // Обновляем заголовок страницы
      if (route.title) {
        document.title = route.title;
      }

      // Сохраняем текущий маршрут
      const previousRoute = this.currentRoute;
      this.currentRoute = route;

      // Уведомляем о завершении навигации
      this.eventBus.emit("route:change", { from: previousRoute, to: route });
      this.eventBus.emit("route:after", { from: previousRoute, to: route });

      return true;
    } catch (error) {
      console.error("Navigation error:", error);
      this.eventBus.emit("route:error", error);
      return false;
    } finally {
      this.isNavigating = false;
    }
  }

  /**
   * Поиск маршрута по пути
   */
  findRoute(path) {
    // Прямое совпадение
    if (this.routes.has(path)) {
      return this.routes.get(path);
    }

    // Поиск с параметрами (упрощенная версия)
    for (const [routePath, route] of this.routes) {
      if (this.matchPath(routePath, path)) {
        return { ...route, params: this.extractParams(routePath, path) };
      }
    }

    return null;
  }

  /**
   * Проверка соответствия пути маршруту
   */
  matchPath(routePath, actualPath) {
    // Простая проверка для примера
    // В реальном приложении используйте более продвинутый алгоритм
    const routeParts = routePath.split("/");
    const actualParts = actualPath.split("/");

    if (routeParts.length !== actualParts.length) {
      return false;
    }

    for (let i = 0; i < routeParts.length; i++) {
      if (routeParts[i].startsWith(":")) {
        continue; // Параметр маршрута
      }
      if (routeParts[i] !== actualParts[i]) {
        return false;
      }
    }

    return true;
  }

  /**
   * Извлечение параметров из пути
   */
  extractParams(routePath, actualPath) {
    const params = {};
    const routeParts = routePath.split("/");
    const actualParts = actualPath.split("/");

    for (let i = 0; i < routeParts.length; i++) {
      if (routeParts[i].startsWith(":")) {
        const paramName = routeParts[i].substring(1);
        params[paramName] = actualParts[i];
      }
    }

    return params;
  }

  /**
   * Проверка доступа к маршруту
   */
  async checkAccess(route) {
    // Публичные маршруты всегда доступны
    if (route.public) {
      return { allowed: true };
    }

    // Получаем состояние авторизации
    const authState = this.getAuthState();

    // Защищенные маршруты требуют авторизации
    if (route.protected && !authState.isAuthenticated) {
      return {
        allowed: false,
        redirectTo: `/login?redirect=${encodeURIComponent(route.path)}`,
      };
    }

    // Проверка ролей
    if (route.roles && route.roles.length > 0) {
      const userRole = authState.user?.role;
      if (!userRole || !route.roles.includes(userRole)) {
        return { allowed: false };
      }
    }

    // Проверка прав доступа
    if (route.permissions && route.permissions.length > 0) {
      const userPermissions = authState.user?.permissions || [];
      const hasPermission = route.permissions.some((permission) =>
        userPermissions.includes(permission)
      );

      if (!hasPermission) {
        return { allowed: false };
      }
    }

    // Запуск пользовательских guard'ов
    for (const [name, guard] of this.guards) {
      try {
        const result = await guard(route, authState);
        if (!result.allowed) {
          return result;
        }
      } catch (error) {
        console.error(`Guard '${name}' failed:`, error);
        return { allowed: false };
      }
    }

    return { allowed: true };
  }

  /**
   * Получение состояния авторизации
   */
  getAuthState() {
    // Должно быть реализовано через DI или глобальное состояние
    if (window.authApp && window.authApp.getState) {
      return window.authApp.getState();
    }

    return {
      isAuthenticated: false,
      user: null,
      permissions: [],
    };
  }

  /**
   * Обработка изменения URL
   */
  handlePopState(event) {
    const path = window.location.pathname;
    this.handleRoute(path);
  }

  /**
   * Обработка текущего маршрута
   */
  async handleCurrentRoute() {
    const path = window.location.pathname;
    await this.handleRoute(path);
  }

  /**
   * Обработка маршрута
   */
  async handleRoute(path) {
    await this.navigate(path, { replace: true });
  }

  /**
   * Возврат назад
   */
  back() {
    history.back();
  }

  /**
   * Переход вперед
   */
  forward() {
    history.forward();
  }

  /**
   * Получение текущего маршрута
   */
  getCurrentRoute() {
    return this.currentRoute;
  }

  /**
   * Получение URL с параметрами
   */
  buildUrl(path, params = {}) {
    let url = path;
    const queryParams = new URLSearchParams();

    for (const [key, value] of Object.entries(params)) {
      if (value !== null && value !== undefined) {
        queryParams.append(key, value.toString());
      }
    }

    const queryString = queryParams.toString();
    if (queryString) {
      url += "?" + queryString;
    }

    return url;
  }

  /**
   * Проверка активного маршрута
   */
  isActive(path) {
    return this.currentRoute && this.currentRoute.path === path;
  }

  /**
   * Получение всех маршрутов
   */
  getRoutes() {
    return Array.from(this.routes.values());
  }

  /**
   * Получение публичных маршрутов
   */
  getPublicRoutes() {
    return this.getRoutes().filter((route) => route.public);
  }

  /**
   * Получение защищенных маршрутов
   */
  getProtectedRoutes() {
    return this.getRoutes().filter((route) => route.protected);
  }
}
```

```javascript
// form-integration.js - Интеграция с формами

/**
 * Класс для интеграции архитектуры с формами HTML
 */
class FormIntegration {
  constructor(authApp) {
    this.authApp = authApp;
    this.activeValidators = new Map();
    this.activeForms = new Map();
  }

  /**
   * Инициализация интеграции с формами
   */
  init() {
    // Автоматически находим и инициализируем формы авторизации
    this.scanAndInitializeForms();

    // Наблюдаем за динамически добавляемыми формами
    this.observeFormChanges();

    console.log("Form integration initialized");
  }

  /**
   * Сканирование и инициализация форм
   */
  scanAndInitializeForms() {
    const authForms = document.querySelectorAll("[data-auth-form]");

    authForms.forEach((form) => {
      this.initializeForm(form);
    });
  }

  /**
   * Инициализация отдельной формы
   */
  initializeForm(form) {
    const formType = form.dataset.authForm;
    const formId = form.id || this.generateFormId();

    if (this.activeForms.has(formId)) {
      console.warn(`Form ${formId} already initialized`);
      return;
    }

    try {
      switch (formType) {
        case "login":
          this.initializeLoginForm(form, formId);
          break;
        case "register":
          this.initializeRegisterForm(form, formId);
          break;
        case "forgot-password":
          this.initializeForgotPasswordForm(form, formId);
          break;
        case "reset-password":
          this.initializeResetPasswordForm(form, formId);
          break;
        default:
          console.warn(`Unknown form type: ${formType}`);
      }
    } catch (error) {
      console.error(`Failed to initialize form ${formId}:`, error);
    }
  }

  /**
   * Инициализация формы входа
   */
  initializeLoginForm(form, formId) {
    const validator = this.authApp.getModule("validator");
    const notifications = this.authApp.getModule("notifications");

    // Настраиваем валидацию
    if (validator) {
      validator.initFormValidation(`#${form.id || formId}`);
      this.activeValidators.set(formId, validator);
    }

    // Обработчик отправки формы
    form.addEventListener("submit", async (e) => {
      e.preventDefault();
      await this.handleLoginSubmit(form, formId);
    });

    // Автозаполнение при восстановлении сессии
    this.authApp.eventBus.on("auth:restored", (data) => {
      if (data.user) {
        this.fillLoginForm(form, data.user);
      }
    });

    this.activeForms.set(formId, {
      element: form,
      type: "login",
      validator: validator,
      initialized: Date.now(),
    });

    console.log(`Login form ${formId} initialized`);
  }

  /**
   * Обработка отправки формы входа
   */
  async handleLoginSubmit(form, formId) {
    const notifications = this.authApp.getModule("notifications");
    const ux = this.authApp.getModule("ux");

    try {
      // Устанавливаем состояние загрузки
      this.setFormLoading(form, true, "Вход в систему...");

      // Собираем данные формы
      const formData = new FormData(form);
      const credentials = {
        loginField: formData.get("loginField") || formData.get("email"),
        password: formData.get("password"),
        rememberMe: formData.get("rememberMe") === "on",
      };

      // Валидируем форму
      const validator = this.activeValidators.get(formId);
      if (validator) {
        const validation = await validator.validateForm(form);
        if (!validation.isValid) {
          if (notifications) {
            notifications.showError("Исправьте ошибки в форме");
          }
          return;
        }
      }

      // Отправляем запрос на авторизацию
      const result = await this.authApp.login(credentials);

      if (result.success) {
        if (notifications) {
          notifications.showSuccess("Вход выполнен успешно!");
        }

        // Перенаправляем пользователя
        this.handleSuccessfulLogin();
      } else {
        if (notifications) {
          notifications.showError(this.authApp.getErrorMessage(result.error));
        }
      }
    } catch (error) {
      console.error("Login submission error:", error);
      if (notifications) {
        notifications.showError("Произошла ошибка при входе");
      }
    } finally {
      this.setFormLoading(form, false);
    }
  }

  /**
   * Инициализация формы регистрации
   */
  initializeRegisterForm(form, formId) {
    const validator = this.authApp.getModule("validator");
    const hints = this.authApp.getModule("hints");

    // Настраиваем валидацию
    if (validator) {
      validator.initFormValidation(`#${form.id || formId}`);

      // Специальная валидация подтверждения пароля
      validator.initPasswordConfirmationValidation(`#${form.id || formId}`);

      this.activeValidators.set(formId, validator);
    }

    // Настраиваем умные подсказки
    if (hints) {
      hints.initFormHints(`#${form.id || formId}`);
    }

    // Обработчик отправки формы
    form.addEventListener("submit", async (e) => {
      e.preventDefault();
      await this.handleRegisterSubmit(form, formId);
    });

    this.activeForms.set(formId, {
      element: form,
      type: "register",
      validator: validator,
      hints: hints,
      initialized: Date.now(),
    });

    console.log(`Register form ${formId} initialized`);
  }

  /**
   * Обработка отправки формы регистрации
   */
  async handleRegisterSubmit(form, formId) {
    const notifications = this.authApp.getModule("notifications");

    try {
      this.setFormLoading(form, true, "Регистрация...");

      const formData = new FormData(form);
      const userData = {
        firstName: formData.get("firstName"),
        lastName: formData.get("lastName"),
        email: formData.get("email"),
        username: formData.get("username"),
        password: formData.get("password"),
        confirmPassword: formData.get("confirmPassword"),
      };

      // Валидируем форму
      const validator = this.activeValidators.get(formId);
      if (validator) {
        const validation = await validator.validateForm(form);
        if (!validation.isValid) {
          if (notifications) {
            notifications.showError("Исправьте ошибки в форме");
          }
          return;
        }
      }

      // Отправляем запрос на регистрацию
      const result = await this.authApp.register(userData);

      if (result.success) {
        if (notifications) {
          notifications.showSuccess("Регистрация прошла успешно!");
        }

        // Автоматический вход или перенаправление
        this.handleSuccessfulRegistration(result.data);
      } else {
        if (notifications) {
          notifications.showError(this.authApp.getErrorMessage(result.error));
        }
      }
    } catch (error) {
      console.error("Registration submission error:", error);
      if (notifications) {
        notifications.showError("Произошла ошибка при регистрации");
      }
    } finally {
      this.setFormLoading(form, false);
    }
  }

  /**
   * Установка состояния загрузки для формы
   */
  setFormLoading(form, loading, message = "") {
    const submitButton = form.querySelector('[type="submit"]');
    const inputs = form.querySelectorAll("input, select, textarea");

    if (submitButton) {
      const buttonText = submitButton.querySelector(".btn-text, .button-text");
      const buttonLoader = submitButton.querySelector(
        ".btn-loader, .button-loader"
      );

      if (loading) {
        if (buttonText) buttonText.style.display = "none";
        if (buttonLoader) {
          buttonLoader.style.display = "flex";
          if (message) {
            const loaderText = buttonLoader.querySelector(".loading-text");
            if (loaderText) loaderText.textContent = message;
          }
        }
        submitButton.disabled = true;
      } else {
        if (buttonText) buttonText.style.display = "block";
        if (buttonLoader) buttonLoader.style.display = "none";
        submitButton.disabled = false;
      }
    }

    // Блокируем/разблокируем поля ввода
    inputs.forEach((input) => {
      input.disabled = loading;
    });

    // Добавляем CSS класс для стилизации
    form.classList.toggle("loading", loading);
  }

  /**
   * Обработка успешного входа
   */
  handleSuccessfulLogin() {
    // Получаем redirect URL из параметров
    const urlParams = new URLSearchParams(window.location.search);
    const redirectUrl = urlParams.get("redirect");

    if (redirectUrl) {
      window.location.href = decodeURIComponent(redirectUrl);
    } else {
      // Перенаправляем на дашборд или главную страницу
      this.authApp.router.navigate("/dashboard");
    }
  }

  /**
   * Обработка успешной регистрации
   */
  handleSuccessfulRegistration(data) {
    if (data.requiresVerification) {
      this.authApp.router.navigate("/verify-email");
    } else {
      // Автоматический вход после регистрации
      this.handleSuccessfulLogin();
    }
  }

  /**
   * Заполнение формы входа сохраненными данными
   */
  fillLoginForm(form, userData) {
    const emailField = form.querySelector(
      '[name="email"], [name="loginField"]'
    );
    if (emailField && userData.email) {
      emailField.value = userData.email;
    }
  }

  /**
   * Наблюдение за изменениями DOM
   */
  observeFormChanges() {
    const observer = new MutationObserver((mutations) => {
      mutations.forEach((mutation) => {
        mutation.addedNodes.forEach((node) => {
          if (node.nodeType === Node.ELEMENT_NODE) {
            // Проверяем добавленный элемент
            if (node.matches && node.matches("[data-auth-form]")) {
              this.initializeForm(node);
            }

            // Проверяем дочерние элементы
            const authForms = node.querySelectorAll("[data-auth-form]");
            authForms.forEach((form) => this.initializeForm(form));
          }
        });
      });
    });

    observer.observe(document.body, {
      childList: true,
      subtree: true,
    });
  }

  /**
   * Генерация ID для формы
   */
  generateFormId() {
    return "auth-form-" + Math.random().toString(36).substring(2, 9);
  }

  /**
   * Получение информации об активных формах
   */
  getActiveForms() {
    return Array.from(this.activeForms.entries()).map(([id, form]) => ({
      id,
      type: form.type,
      initialized: form.initialized,
      element: form.element,
    }));
  }

  /**
   * Уничтожение формы
   */
  destroyForm(formId) {
    const formData = this.activeForms.get(formId);

    if (formData) {
      // Очищаем валидаторы
      if (formData.validator) {
        formData.validator.clearValidation(`#${formId}`);
      }

      // Удаляем обработчики событий (если необходимо)

      this.activeForms.delete(formId);
      this.activeValidators.delete(formId);

      console.log(`Form ${formId} destroyed`);
    }
  }

  /**
   * Очистка всех форм
   */
  cleanup() {
    this.activeForms.forEach((_, formId) => {
      this.destroyForm(formId);
    });
  }
}
```

### 2.4 Основной файл инициализации

```javascript
// main.js - Основной файл инициализации

/**
 * Глобальная инициализация приложения авторизации
 */
(function () {
  "use strict";

  // Конфигурация приложения
  const config = {
    apiBaseUrl: "/api",
    tokenStorageKey: "authToken",
    userStorageKey: "currentUser",
    refreshThreshold: 5 * 60 * 1000, // 5 минут
    maxRetries: 3,
    retryDelay: 1000,
    timeout: 30000,
    debug: true,
  };

  // Глобальная переменная для приложения
  window.authApp = null;

  /**
   * Инициализация приложения при загрузке DOM
   */
  async function initializeApp() {
    try {
      console.log("Initializing Auth Application...");

      // Создаем экземпляр приложения
      window.authApp = new AuthApplication(config);

      // Инициализируем приложение
      await window.authApp.init();

      // Инициализируем интеграцию с формами
      const formIntegration = new FormIntegration(window.authApp);
      formIntegration.init();

      // Настраиваем глобальные обработчики
      setupGlobalHandlers();

      // Настраиваем обработчики ошибок
      setupErrorHandlers();

      console.log("Auth Application initialized successfully");

      // Уведомляем о готовности приложения
      window.dispatchEvent(
        new CustomEvent("authAppReady", {
          detail: { app: window.authApp },
        })
      );
    } catch (error) {
      console.error("Failed to initialize Auth Application:", error);

      // Показываем пользователю сообщение об ошибке
      showInitializationError(error);
    }
  }

  /**
   * Настройка глобальных обработчиков
   */
  function setupGlobalHandlers() {
    // Обработчик для логаута при закрытии вкладки
    window.addEventListener("beforeunload", () => {
      if (window.authApp) {
        const state = window.authApp.getState();
        if (state.isAuthenticated && !state.remember) {
          // Только для сессионной авторизации
          window.authApp.logout();
        }
      }
    });

    // Обработчик для перехвата ссылок на защищенные ресурсы
    document.addEventListener("click", (e) => {
      const link = e.target.closest("a[data-protected]");
      if (link && window.authApp) {
        const state = window.authApp.getState();
        if (!state.isAuthenticated) {
          e.preventDefault();
          const href = link.getAttribute("href");
          window.authApp.router.navigate(
            `/login?redirect=${encodeURIComponent(href)}`
          );
        }
      }
    });

    // Обработчик для автоматического обновления UI при изменении состояния
    if (window.authApp) {
      window.authApp.eventBus.on(
        "state:authenticated",
        updateUIForAuthenticatedUser
      );
      window.authApp.eventBus.on(
        "state:unauthenticated",
        updateUIForUnauthenticatedUser
      );
    }
  }

  /**
   * Настройка обработчиков ошибок
   */
  function setupErrorHandlers() {
    // Глобальный обработчик необработанных ошибок
    window.addEventListener("error", (e) => {
      console.error("Global error:", e.error);

      if (window.authApp) {
        const notifications = window.authApp.getModule("notifications");
        if (notifications && config.debug) {
          notifications.showError("Произошла неожиданная ошибка");
        }
      }
    });

    // Обработчик необработанных промисов
    window.addEventListener("unhandledrejection", (e) => {
      console.error("Unhandled promise rejection:", e.reason);

      if (window.authApp) {
        const notifications = window.authApp.getModule("notifications");
        if (notifications && config.debug) {
          notifications.showError("Ошибка в асинхронной операции");
        }
      }
    });
  }

  /**
   * Обновление UI для авторизованного пользователя
   */
  function updateUIForAuthenticatedUser(state) {
    // Показываем элементы для авторизованных пользователей
    document
      .querySelectorAll('[data-auth-show="authenticated"]')
      .forEach((el) => {
        el.style.display = "";
      });

    // Скрываем элементы для неавторизованных пользователей
    document
      .querySelectorAll('[data-auth-show="unauthenticated"]')
      .forEach((el) => {
        el.style.display = "none";
      });

    // Обновляем информацию о пользователе
    document.querySelectorAll("[data-user-info]").forEach((el) => {
      const field = el.dataset.userInfo;
      if (state.user && state.user[field]) {
        el.textContent = state.user[field];
      }
    });
  }

  /**
   * Обновление UI для неавторизованного пользователя
   */
  function updateUIForUnauthenticatedUser() {
    // Скрываем элементы для авторизованных пользователей
    document
      .querySelectorAll('[data-auth-show="authenticated"]')
      .forEach((el) => {
        el.style.display = "none";
      });

    // Показываем элементы для неавторизованных пользователей
    document
      .querySelectorAll('[data-auth-show="unauthenticated"]')
      .forEach((el) => {
        el.style.display = "";
      });

    // Очищаем информацию о пользователе
    document.querySelectorAll("[data-user-info]").forEach((el) => {
      el.textContent = "";
    });
  }

  /**
   * Показ ошибки инициализации
   */
  function showInitializationError(error) {
    const errorContainer = document.createElement("div");
    errorContainer.className = "initialization-error";
    errorContainer.innerHTML = `
            <div class="error-content">
                <h3>Ошибка инициализации</h3>
                <p>Не удалось запустить систему авторизации.</p>
                <details>
                    <summary>Подробности</summary>
                    <pre>${error.message}</pre>
                </details>
                <button onclick="location.reload()">Перезагрузить</button>
            </div>
        `;

    document.body.appendChild(errorContainer);
  }

  /**
   * Публичный API для внешнего использования
   */
  window.AuthApp = {
    // Получение экземпляра приложения
    getInstance: () => window.authApp,

    // Проверка готовности приложения
    isReady: () => window.authApp && window.authApp.isInitialized,

    // Ожидание готовности приложения
    ready: () => {
      return new Promise((resolve) => {
        if (window.AuthApp.isReady()) {
          resolve(window.authApp);
        } else {
          window.addEventListener(
            "authAppReady",
            (e) => {
              resolve(e.detail.app);
            },
            { once: true }
          );
        }
      });
    },

    // Удобные методы для быстрого доступа
    login: async (credentials) => {
      const app = await window.AuthApp.ready();
      return app.login(credentials);
    },

    register: async (userData) => {
      const app = await window.AuthApp.ready();
      return app.register(userData);
    },

    logout: async () => {
      const app = await window.AuthApp.ready();
      return app.logout();
    },

    getState: async () => {
      const app = await window.AuthApp.ready();
      return app.getState();
    },
  };

  // Запускаем инициализацию при загрузке DOM
  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", initializeApp);
  } else {
    // DOM уже загружен
    initializeApp();
  }
})();
```
